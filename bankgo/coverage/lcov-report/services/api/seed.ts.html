function buildTransactions(
  accountId: string,
  items: Omit<Transaction, 'id' | 'accountId'>[],
): Transaction[] {

    id: `tx-${accountId}-${String(i + 1).padStart(3, '0')}`,
    accountId,
    ...item,
  }));
}
